import streamlit as st

# -----------------------------
# Page configuration
# -----------------------------

st.set_page_config(
    page_title="Binary Number System Converter",
    page_icon="🔢",
    layout="centered"
)

# -----------------------------
# Title
# -----------------------------

st.title("🔢 Binary Number System Converter")
st.write("Convert numbers between Binary, Octal, Decimal and Hexadecimal.")

# -----------------------------
# Number system bases
# -----------------------------

BASES = {
    "Binary": 2,
    "Octal": 8,
    "Decimal": 10,
    "Hexadecimal": 16
}

# -----------------------------
# Validation function
# -----------------------------

def validate_number(number, base):

    number = number.strip().upper()

    valid_digits = {
        2: "01",
        8: "01234567",
        10: "0123456789",
        16: "0123456789ABCDEF"
    }

    if number == "":
        return False

    return all(char in valid_digits[base] for char in number)


# -----------------------------
# Conversion function
# -----------------------------

def convert_number(number, from_base, to_base):

    decimal_number = int(number, from_base)

    if to_base == 2:
        return bin(decimal_number)[2:]

    elif to_base == 8:
        return oct(decimal_number)[2:]

    elif to_base == 10:
        return str(decimal_number)

    elif to_base == 16:
        return hex(decimal_number)[2:].upper()


# -----------------------------
# User input
# -----------------------------

number = st.text_input(
    "Enter Number",
    placeholder="Example: 1010"
)

from_system = st.selectbox(
    "Convert From",
    ["Binary", "Octal", "Decimal", "Hexadecimal"]
)

to_system = st.selectbox(
    "Convert To",
    ["Binary", "Octal", "Decimal", "Hexadecimal"]
)

# -----------------------------
# Convert button
# -----------------------------

if st.button("🔄 Convert", use_container_width=True):

    if number.strip() == "":
        st.error("Please enter a number.")

    else:

        from_base = BASES[from_system]
        to_base = BASES[to_system]

        if not validate_number(number, from_base):

            st.error(
                f"Invalid {from_system} number. "
                "Please check your input."
            )

        else:

            result = convert_number(
                number.upper(),
                from_base,
                to_base
            )

            st.success("Conversion Successful!")

            st.subheader("Result")

            st.code(result, language=None)

# -----------------------------
# Information
# -----------------------------

st.divider()

st.subheader("Number Systems")

st.write("""
- **Binary:** Base 2 → digits 0 and 1
- **Octal:** Base 8 → digits 0 to 7
- **Decimal:** Base 10 → digits 0 to 9
- **Hexadecimal:** Base 16 → digits 0 to 9 and A to F
""")
